  static mount(containerEl, simConfig) {
    switch (simConfig.type) {
      case 'orbit3d': return mountOrbit3DSolarSystemEngine(containerEl, simConfig);
      case 'pendulum': return mountPendulumEngine(containerEl, simConfig);
      case 'circuit': return mountCircuitElectronEngine(containerEl, simConfig);
      default: return mountOrbit3DSolarSystemEngine(containerEl, simConfig);
    }
  }
